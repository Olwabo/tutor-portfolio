import java.util.Arrays;
public class MyArray {
    public void printArray(int[] array, int size)
    {
        for(int i = 0; i < size; i++){
            System.out.print(array[i] + " ");
        }
    }
    public int searchArray(int[] array, int size, int item){
        boolean found = false;
        int i;
        for( i = 0; i < size; i++){
            if(array[i] == item){
                found = true;
                break;
            }
        }
        if(found)
            return i;
        else
            return -1;
    }
    public int minimum(int[] array, int size){
        int min = array[0];
        for(int i = 0; i < size; i++){
            if(min > array[i]){
                min = array[i];
            }
        }
        return min;
    }
    public int maximum(int[] array, int size){
        int max = array[0];
        for(int i = 0; i < size; i++){
            if(max < array[i]){
                max = array[i];
            }
        }
        return max;
    }
    public int sumArray(int[] array, int size){
        int sum = 0;
        for(int i = 0; i < size; i++){
            sum = sum + array[i];
        }
        return sum;
    }
    public double averageArray(int[] array, int size){
        double sum = 0.0, average;
        for(int i = 0; i < size; i++){
            sum = sum + array[i];
        }
        average = sum / size;
        return average;
    }
    public void reverseArray(int[] array, int size){
        for(int i = size - 1; i >= 0; i--){
            System.out.println(array[i]);
        }
    }
}
